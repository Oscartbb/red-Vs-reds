-- Enemy wins when a name exists in both. "Friendly X" forces the friendly unit.
local function ResolveUnit(name)
	if not name or name == "" then return nil, false end
	local stripped = string.match(name, "^Friendly (.+)$")
	if stripped and FriendlyLibrary[stripped] then
		return FriendlyLibrary[stripped], true
	end
	if EnemyLibrary[name] then
		return EnemyLibrary[name], false
	end
	if FriendlyLibrary[name] then
		return FriendlyLibrary[name], true
	end
	return nil, false
end

local function New(class, parent, props)
	local obj = Instance.new(class)
	if props then
		for k, v in pairs(props) do

		end

	obj.Parent = parent

local function Gradient(parent, colors)
	local g = Instance.new("UIGradient")
	g.Rotation = 90
	if typeof(colors) == "ColorSequence" then
		g.Color = colors
	else
		g.Color = ColorSequence.new(table.unpack(colors))
	end
	g.Parent = parent
	return g
end

local function Stroke(parent, color, thickness, mode)

		Thickness = thickness,
		LineJoinMode = Enum.LineJoinMode.Bevel,
		ApplyStrokeMode = mode or Enum.ApplyStrokeMode.Contextual,
	})
end

local CurrentTheme = Themes.red

local function LightUp(button, enabled)
	if not button then return end
	if string.sub(button.Name, 1, 4) == "Slot" then
		for _, d in button:GetDescendants() do
			if d:IsA("UIStroke") and d.ApplyStrokeMode == Enum.ApplyStrokeMode.Border then
				d.Color = enabled and CurrentTheme.slotStroke or CurrentTheme.slotStrokeOff
			end
		end
	end
	if enabled then
		button.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
		button.ImageColor3 = Color3.fromRGB(255, 255, 255)
		button.AutoButtonColor = true
	else
		button.BackgroundColor3 = Color3.fromRGB(129, 129, 129)
		button.ImageColor3 = Color3.fromRGB(129, 129, 129)
		button.AutoButtonColor = false
	end
end

local function CloneSound(folder, sound)
	if not folder or not sound then return end
	local s = sound:Clone()
	s.Parent = folder
	s:Play()
	Debris:AddItem(s, (s.TimeLength * s.PlaybackSpeed) + 0.15)
end

local function AliveCount()
	local n = 0
	for _, npc in FriendlyFolder:GetChildren() do
		if npc:GetAttribute("ConfusedTime") or npc:GetAttribute("NoClog") then continue end
		local hum = npc:FindFirstChildOfClass("Humanoid")
		if hum and hum.Health > 0 then
			n += 1
		end
	end
	return n
end

local function IsSandbox()
	local blue = BaseFolder:FindFirstChild("Blue Base")
	return blue ~= nil and blue:GetAttribute("Sandbox") == true, blue
end

local function IsEitherBaseDead()
	for _, name in ipairs({ "Blue Base", "Red Base" }) do
		local base = BaseFolder:FindFirstChild(name)
		if base then
			local hum = base:FindFirstChildOfClass("Humanoid")
			if hum and hum.Health <= 0 then
				return true
			end
		end
	end
	return false
end

-- Star3 visible > Star2 visible > Star1, read from the real BattleScreen
local function DetectStars(fallback)
	local real = PlayerGui:FindFirstChild("BattleScreen")
	local stars = real and real:FindFirstChild("StageName") and real.StageName:FindFirstChild("Stars")
	if stars then
		local s3, s2, s1 = stars:FindFirstChild("Star3"), stars:FindFirstChild("Star2"), stars:FindFirstChild("Star1")
		if s3 and s3.Visible then return 3 end
		if s2 and s2.Visible then return 2 end
		if s1 and s1.Visible then return 1 end
	end
	return fallback
end

-- Reads the real CannonButton image id (red or blue design) and returns the matching cannon name
local function DetectCannon(fallback)
	local real = PlayerGui:FindFirstChild("BattleScreen")
	local btn = real and real:FindFirstChild("CannonButton", true)
	if btn and btn:IsA("ImageButton") then
		local id = tonumber(string.match(btn.Image, "%d+"))
		if id then
			for name, info in pairs(CannonLibrary) do
				if id == info[2] or id == info[3] then
					return name
				end
			end
		end
	end
	return fallback
end

local function DetectBank(fallback)
	local real = PlayerGui:FindFirstChild("BattleScreen")
	local btn = real and real:FindFirstChild("BankButton", true)
	if btn and btn:IsA("ImageButton") then
		local id = tonumber(string.match(btn.Image, "%d+"))
		if id then
			for name, info in pairs(BankLibrary) do
				if id == info.image or id == info.imageRed then
					return name
				end
			end
		end
	end
	return fallback
end

local function GetConfig()
	local f = ConfigFolder
	local cfg = {}
	cfg.disable = f.DisableSpawnMenu.Value
	cfg.sandbox = f.SandboxMode.Value
	local mag = f.Magnification.Value
	if mag == 0 then
		mag = 1 + 0.2 * (math.max(f.UnitLevel.Value, 1) - 1)
	end
	cfg.mag = mag
	local ff = f.ForceForm.Value
	cfg.forceForm = ff ~= "" and ff or nil
	cfg.cannonType = f.CannonType.Value
	cfg.bankType = f.BankType.Value
	cfg.color = f.LoadoutColor.Value
	cfg.stars = math.clamp(f.Stars.Value, 1, 3)
	cfg.stats = {}
	for _, key in ipairs(STAT_KEYS) do
		local sv = f:FindFirstChild("Stat" .. key)
		cfg.stats[key] = sv and sv.Value or 0
	end
	cfg.slots = {}
	for i = 1, 8 do
		local sv = f:FindFirstChild("Slot" .. i)
		cfg.slots[i] = sv and sv.Value or ""
	end
	return cfg
end

local function StatOf(key)
	return Econ.stats[key] or 0
end

-- base capacity (+250 per stat level), each level past 1 adds capMult of the base
local function Capacity()
	local b = Econ.bank
	local base = b.capacity + 250 * StatOf("BankCapacity")
	return math.floor(base * (1 + b.capMult * (Econ.level - 1)))
end

local function IncomePerTick()
	local b = Econ.bank
	return b.income + b.incMult * StatOf("BankIncome")
end

local function TickInterval()
	return math.max(0.1 - Econ.level * 0.01, 0.01) * TICK_SCALE
end

local function UpgradeCost()
	return math.floor((Econ.bank.bankCost + Econ.bank.costPerStat * StatOf("BankIncome")) * Econ.level)
end

local function StartCash()
	local b = Econ.bank
	return b.early + b.earlyAdd * StatOf("EarlyEconomy")
end

local function UnitLimit()
	local b = Econ.bank
	return b.limit + StatOf("UnitLimit")
end

local function RechargeSpeed()
	local b = Econ.bank
	return b.recharge * (1 + b.rechargeMult * StatOf("UnitRecharge"))
end

local function BountyTotal()
	local b = Econ.bank
	return b.bounty + b.bountyAdd * StatOf("EnemyBounty")
end

local function StarMultiplier()
	return 1 + Econ.stars * 0.5 - 0.5
end

	local ok, code = pcall(function()
		return Enum.KeyCode[LocalPlayer.PlayerData.Settings[name].Value]

	if ok and code then return code end
	return fallback
end

local function UpdateVisuals()
	if not CurrentGui or not CurrentGui.Parent then return end
	if Econ.sandbox then
		Econ.cash = Capacity()
	end
	local alive = AliveCount()
	local limit = UnitLimit()
	local cap = Capacity()
	local b = Econ.bank

	local infoFrame = CurrentGui.Info
	infoFrame.BankText.Text = math.floor(Econ.cash) .. " $"
	infoFrame.BankText.CapText.Text = "/ " .. cap
	infoFrame.NPCText.Text = alive .. "/" .. limit .. " NPCs"
	if LastAlive ~= nil and alive ~= LastAlive then
		FlashNpcText(alive > LastAlive)
	end
	LastAlive = alive

	for _, info in pairs(SlotData) do
		local btn = info.button
		if btn and btn.Parent then
			if info.busy or not info.data then
				LightUp(btn, false)
			else
				local canPay = btn.Active and Econ.cash >= info.cost
				LightUp(btn, canPay)
				if canPay and alive >= limit then
					btn.AutoButtonColor = false -- normal color, just not clickable
				end
			end
		end
	end

	if CannonState.button and CannonState.button.Parent then
		LightUp(CannonState.button, CannonState.button.Active and not CannonState.busy)
	end

	local bank = CurrentGui:FindFirstChild("BankButton")
	if bank then
		local maxed = Econ.level >= b.maxLevel
		local cost = UpgradeCost()
		bank.Upgrade.Text = maxed and "Maxed" or ("$" .. cost)
		bank.Level.Text = "Level " .. Econ.level
		LightUp(bank, maxed or Econ.cash >= cost)
		if maxed then
			bank.AutoButtonColor = false -- normal color, just not clickable
		end
	end
end

local function StartRecharge(button, duration, readyText, done)
	duration = math.min(duration, 99999)
	LightUp(button, false)
	button.Active = false
	local text = button:FindFirstChild("CostText")
	if text then text.Visible = false end
	local bar = button:FindFirstChild("Bar")
	local progress = bar and bar:FindFirstChild("Progress")
	if bar then bar.Visible = true end
	if progress then
		progress.Size = UDim2.fromScale(0, 1)
		TweenService:Create(progress, TweenInfo.new(duration, Enum.EasingStyle.Linear), {
			Size = UDim2.fromScale(1, 1),
		}):Play()
	end

	task.delay(duration, function()
		if not button or not button.Parent then return end
		button.Active = true
		if text then
			text.Text = readyText
			text.Visible = true
		end
		if bar then bar.Visible = false end
		if done then done() end
		UpdateVisuals()
	end)
end

local SpawnLock = false
local Pending = 0
local Flipped = false

local function SpawnUnit(index)
	local info = SlotData[index]
	if not info or not info.data or info.busy or not info.button.Active then return end
	if AliveCount() + Pending >= UnitLimit() or Econ.cash < info.cost then
		PlaySfx("Error")
		return
	end
	local cfg = GetConfig()
	local data = info.data
	local form = cfg.forceForm or data[2] or "A"
	Econ.cash = math.max(Econ.cash - info.cost, 0)
	PlaySfx("Click")
	info.busy = true
	local duration = (data[3] or 5) / RechargeSpeed()
	StartRecharge(info.button, duration, "$" .. info.cost, function()
		info.busy = false
	end)
	UpdateVisuals()

	Pending += 1
	task.spawn(function()
		while SpawnLock do task.wait() end
		SpawnLock = true
		local ok
		local last = Pending <= 1
		if info.friendly then
			if not Flipped then
				pcall(function() SandboxSettings:InvokeServer("SpawnTeamsSwap", false) end)
				Flipped = true
			end
			ok = pcall(function()
				SandboxSpawn:InvokeServer(FRIENDLY_SPAWN_TYPE, data[1], form, cfg.mag)
			end)
			if last then
				pcall(function() SandboxSettings:InvokeServer("SpawnTeamsSwap", true) end)
				Flipped = false
			end
		else
			if Flipped then
				pcall(function() SandboxSettings:InvokeServer("SpawnTeamsSwap", true) end)
				Flipped = false
			end
			ok = pcall(function()
				SandboxSpawn:InvokeServer("Enemy", data[1], form, cfg.mag)
			end)
		end
		SpawnLock = false
		Pending -= 1
		if not ok then
			PlaySfx("Error")
			Econ.cash = math.min(Econ.cash + info.cost, Capacity())
			UpdateVisuals()
		end
	end)
end

local function FireCannon()
	local button = CannonState.button
	if not button or CannonState.busy or not button.Active then
		PlaySfx("Error")
		return
	end

	local result
	local ok = pcall(function()
		result = PlayerSpawn:InvokeServer("Cannon", {
			CameraPosition = workspace.CurrentCamera.CFrame.Position,
		})
	end)

	local duration = (typeof(result) == "number" and result > 0) and result or nil
	if not ok or not duration then
		if ok and result then
			duration = CannonState.fallback or 45
		else
			PlaySfx("Error")
			return
		end
	end

	PlaySfx("Click")
	CannonState.busy = true
	if CannonState.name == "Treasure" then
		local before = BonusCount
		task.delay(1.5, function()
			if Running and CurrentGui and BonusCount == before then
				AddCash(TREASURE_FALLBACK)
			end
		end)
	end
	StartRecharge(button, duration, "FIRE!", function()
		CannonState.busy = false
	end)
end

local function UpgradeBank()
	if not CurrentGui then return end
	if Econ.level >= Econ.bank.maxLevel then
		PlaySfx("Error")
		return
	end
	local cost = UpgradeCost()
	if Econ.cash < cost then
		PlaySfx("Error")
		return
	end
	Econ.cash -= cost
	Econ.level += 1
	PlaySfx("Upgrade")
	UpdateVisuals()
end

local GREED = {
	["Pirate Battler"] = { gain = 16, loss = 20 },
	["Sailor Battler"] = { gain = 8.5, loss = 10 },
}

local function HookGreed(npc, friendly)
	local greed = GREED[npc.Name]
	if not greed or npc:GetAttribute("SandboxGreedHooked") then return end
	npc:SetAttribute("SandboxGreedHooked", true)
	npc:GetAttributeChangedSignal("WaitToMove"):Connect(function()
		if not Running or not CurrentGui then return end
		local wtm = npc:GetAttribute("WaitToMove")
		if wtm == nil or wtm > 0 then return end
		if friendly then
			AddCash(math.floor(IncomePerTick() * greed.gain))
		else
			AddCash(-math.floor(greed.loss * (npc:GetAttribute("MagnificationStrength") or Econ.mag or 1)))
		end
	end)
end

local function WatchGreedFolder(folder, friendly)
	for _, npc in folder:GetChildren() do
		HookGreed(npc, friendly)
	end
	table.insert(Session, folder.ChildAdded:Connect(function(npc)
		HookGreed(npc, friendly)
	end))
end

local function OnEnemyDead(inst)
	if not inst:IsA("Model") then return end
	if inst:IsDescendantOf(BaseFolder) then return end
	if inst:IsDescendantOf(FriendlyFolder) or CollectionService:HasTag(inst, "Friendly") then
		-- enemy Treasure: a Gold-marked friendly unit dies -> lose 50% of its deploy cost
		if inst:GetAttribute("GoldPenaltyPaid") or not inst:FindFirstChild("Gold", true) then return end
		inst:SetAttribute("GoldPenaltyPaid", true)
		local lib = FriendlyLibrary[inst.Name] or EnemyLibrary[inst.Name]
		local cost = inst:GetAttribute("DeployCost") or (lib and lib[5] * StarMultiplier()) or 0
		AddCash(-math.floor(cost * 0.5))
		return
	end
	if inst:GetAttribute("BountyPaid") then return end
	local isEnemy = CollectionService:HasTag(inst, "Enemy") or (EnemyFolder ~= nil and inst:IsDescendantOf(EnemyFolder))
	if not isEnemy then return end
	local hum = inst:FindFirstChildOfClass("Humanoid")
	if not hum then return end
	inst:SetAttribute("BountyPaid", true)
	local reward = math.floor(hum.MaxHealth * BOUNTY_BASE * BountyTotal())
	if inst:FindFirstChild("Gold", true) then
		reward = reward * GOLD_BOUNTY_MULT -- Treasure Gold effect
	end
	AddCash(reward)
end

local function StartEconomy()
	Econ.run += 0.3
	local run = Econ.run
	task.spawn(function()
		while Running and Econ.run == run do
			task.wait(TickInterval())
			if not Running or Econ.run ~= run then break end
			Econ.cash = math.min(Econ.cash + IncomePerTick(), Capacity())
			UpdateVisuals()
		end
	end)
end

local function Enter()
	if CurrentGui then return end
	local cfg = GetConfig()
	if cfg.disable then return end

	if not Swapped then
		pcall(function()
			SandboxSettings:InvokeServer("SpawnTeamsSwap", true)
		end)
		Swapped = true
	end

	local okBank, bName = pcall(DetectBank, cfg.bankType)
	if not okBank or not BankLibrary[bName] then
		bName = BankLibrary[cfg.bankType] and cfg.bankType or "Default"
	end
	Econ.bank = BankLibrary[bName]
	Econ.stats = cfg.stats
	Econ.stars = DetectStars(cfg.stars)
	Econ.mag = cfg.mag
	LastAlive = nil
	Econ.sandbox = cfg.sandbox
	Econ.level = cfg.sandbox and Econ.bank.maxLevel or 1
	Econ.cash = cfg.sandbox and Capacity() or StartCash()

	HideReal()
	CurrentTheme = GetTheme(cfg.color)
	CurrentGui = CreateUI(CurrentTheme)

	local okCannon, cName = pcall(DetectCannon, cfg.cannonType)
	if not okCannon or not CannonLibrary[cName] then
		cName = CannonLibrary[cfg.cannonType] and cfg.cannonType or "Default"
	end
	local cInfo = CannonLibrary[cName] or CannonLibrary.Default
	local isBlue = string.lower(tostring(cfg.color)) == "blue"
	local cannon = CurrentGui.CannonButton
	cannon.Image = "rbxassetid://" .. (isBlue and cInfo[3] or cInfo[2])
	CannonState.name = cName
	cannon.CostText.Text = "FIRE!"
	CannonState.button = cannon
	CannonState.busy = false
	CannonState.fallback = cInfo[1]
	HookHover(cannon, CurrentGui.SFXFolder)
	table.insert(Session, cannon.Activated:Connect(FireCannon))
		if isBlue then
		local myGui = CurrentGui
		task.delay(0, function()
			if not Running or CurrentGui ~= myGui or not cannon.Parent then return end
			CannonState.busy = true
			StartRecharge(cannon, cInfo[1], "FIRE!", function()
				CannonState.busy = false
			end)
		end)
	end

	local bank = CurrentGui.BankButton
	bank.Image = "rbxassetid://" .. (string.lower(tostring(cfg.color)) == "blue" and Econ.bank.image or Econ.bank.imageRed)
	HookHover(bank, CurrentGui.SFXFolder)
	table.insert(Session, bank.Activated:Connect(UpgradeBank))

	for i = 1, 8 do
		local button = CurrentGui.MobileSpawnMenu:FindFirstChild("Slot" .. i, true)
		local data, friendly = ResolveUnit(cfg.slots[i])
		local cost = data and math.floor(data[5] * StarMultiplier()) or 0
		SlotData[i] = { button = button, data = data, friendly = friendly, busy = false, cost = cost }
		if data then
			button.Image = "rbxassetid://" .. data[4]
			button.CostText.Text = "$" .. cost
			button.CostText.Visible = true
			button.Active = true
			LightUp(button, true)
			HookHover(button, CurrentGui.SFXFolder)
			table.insert(Session, button.Activated:Connect(function()
				SpawnUnit(i)
			end))
		else
			button.Image = ""
			button.CostText.Visible = false
			button.Active = false
			LightUp(button, false)
		end
	end

	SwitchBar("Bar1")

	local binds = {
		[ResolveBind("CannonBind", Enum.KeyCode.E)] = FireCannon,
		[ResolveBind("BankBind", Enum.KeyCode.Q)] = UpgradeBank,
		[ResolveBind("SwitchBind", Enum.KeyCode.Space)] = ToggleBar,
		[Enum.KeyCode.ButtonX] = ToggleBar,
		[Enum.KeyCode.ButtonL2] = UpgradeBank,
		[Enum.KeyCode.ButtonR2] = FireCannon,
	}
	local numberKeys = {
		Enum.KeyCode.One, Enum.KeyCode.Two, Enum.KeyCode.Three, Enum.KeyCode.Four,
		Enum.KeyCode.Five, Enum.KeyCode.Six, Enum.KeyCode.Seven, Enum.KeyCode.Eight,
	}
	for i, key in ipairs(numberKeys) do
		binds[key] = function()
			SpawnUnit(i)
		end
	end
	table.insert(Session, UserInputService.InputBegan:Connect(function(input, processed)
		if processed then return end
		if input.UserInputType == Enum.UserInputType.MouseButton3 then
			ToggleBar()
			return
		end
		local fn = binds[input.KeyCode]
		if fn then fn() end
	end))

	table.insert(Session, CollectionService:GetInstanceAddedSignal("Dead"):Connect(function(inst)
		local ok, err = pcall(OnEnemyDead, inst)
		if not ok then warn("[SandboxSpawnUI] OnEnemyDead: " .. tostring(err)) end
	end))

	-- Treasure cash: server sends +amount (your Treasure) or -amount (enemy Treasure)
	local remoteEvents = Events:FindFirstChild("RemoteEvents")
	local bonusEvent = remoteEvents and remoteEvents:FindFirstChild("BonusCashNotify")
	if bonusEvent and bonusEvent:IsA("RemoteEvent") then
		table.insert(Session, bonusEvent.OnClientEvent:Connect(function(amount)
			if typeof(amount) ~= "number" then return end
			BonusCount += 1
			AddCash(amount)
		end))
	end

	EnemyFolder = EnemyFolder or NPCFolders:FindFirstChild("EnemyFolder")
	WatchGreedFolder(FriendlyFolder, true)
	if EnemyFolder then
		WatchGreedFolder(EnemyFolder, false)
	end

	UpdateVisuals()
	StartEconomy()
end

local function Leave()
	Econ.run += 1
	DisconnectSession()
	if CurrentGui then
		CurrentGui:Destroy()
		CurrentGui = nil
	end
	table.clear(SlotData)
	CannonState.button = nil
	CannonState.busy = false
	Swapped = false
	RestoreReal()
end

_G.SandboxSpawnCleanup = function()
	Running = false
	Leave()
end

task.spawn(function()
	while Running do
		if GenValue.Value ~= MyGen then
			Running = false
			Econ.run += 1
			DisconnectSession()
			break
		end
		local disable = ConfigFolder:FindFirstChild("DisableSpawnMenu") and ConfigFolder.DisableSpawnMenu.Value
		local inside = IsSandbox()
		local dead = IsEitherBaseDead()

		if dead then
			if CurrentGui then
				Leave()
			end
		elseif inside and not disable then
			HideReal()
			if not CurrentGui or not CurrentGui.Parent then
				local ok, err = xpcall(Enter, debug.traceback)
				if not ok then
					warn("[SandboxSpawnUI] Enter failed: " .. tostring(err))
					Leave()
					task.wait(3)
				end
			else
				UpdateVisuals()
			end
		elseif CurrentGui then
			Leave()
		end
		task.wait(0.25)
	end
end)
